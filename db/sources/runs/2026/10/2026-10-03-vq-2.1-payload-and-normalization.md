_original = socket.getaddrinfo
_lock = threading.Lock()
_cache = {}
_log = Path('.build/quantization-research/vq-2.1-scoped-dns-v2.jsonl')


def resolve(host, port, family=0, type=0, proto=0, flags=0):
    name = host.decode('ascii') if isinstance(host, bytes) else host
    admitted = isinstance(name, str) and (name == 'huggingface.co' or name.endswith('.huggingface.co') or name.endswith('.hf.co'))
    if not admitted:
        return _original(host, port, family, type, proto, flags)
    if family not in (socket.AF_UNSPEC, socket.AF_INET):
        raise socket.gaierror(socket.EAI_FAMILY, 'research resolver admits IPv4 only')
    with _lock:
        now=time.monotonic()
        if name not in _cache or _cache[name][0] <= now:
            url='https://1.1.1.1/dns-query?' + urllib.parse.urlencode({'name':name,'type':'A'})
            request=urllib.request.Request(url,headers={'Accept':'application/dns-json'})
            # Public resolver addressed literally, with ordinary verified TLS.
            opener=urllib.request.build_opener(urllib.request.ProxyHandler({}))
            with opener.open(request,timeout=20) as response:raw=response.read(65537)
            if len(raw)>65536:raise socket.gaierror(socket.EAI_FAIL,'DNS response exceeds its bound')
            result=json.loads(raw)

            for answer in result.get('Answer',[]):
                if answer.get('type') != 1:continue
                address=ipaddress.IPv4Address(answer['data'])
                if not address.is_global:raise socket.gaierror(socket.EAI_FAIL,'Non-public artifact address refused')
                addresses.append(str(address));ttl=min(ttl,max(1,int(answer.get('TTL',1))))

            with _log.open('a') as out:out.write(json.dumps({'host':name,'resolver':'https://1.1.1.1/dns-query','addresses':addresses,'ttl_seconds':ttl,'seconds':time.monotonic()-now})+'\n')
        addresses=_cache[name][1]
    return [entry for address in addresses for entry in _original(address,port,socket.AF_INET,type,proto,flags|socket.AI_NUMERICHOST)]
